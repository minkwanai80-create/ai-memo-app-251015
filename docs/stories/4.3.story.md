## Story

**As a** 사용자,
**I want** 노트 내용을 분석하여 AI가 관련 태그를 자동으로 생성해주는 기능,
**so that** 노트를 쉽게 분류하고 검색할 수 있다.

## Acceptance Criteria

1. 노트 상세 페이지에 "AI 태그" 버튼이 표시되어야 한다
2. 버튼 클릭 시 Gemini API로 노트 내용이 전송되어야 한다
3. 최대 6개의 관련 태그가 생성되어야 한다
4. 태그는 한글 또는 영문 단어로 생성되어야 한다
5. 생성된 태그가 노트에 자동으로 추가되어야 한다
6. 태그는 쉼표로 구분하여 DB에 저장되어야 한다
7. 이미 태그가 있는 노트는 태그를 표시해야 한다
8. 중복 태그는 자동으로 제거되어야 한다
9. 태그 생성 실패 시 에러 메시지가 표시되어야 한다
10. 태그는 수동으로 추가/삭제할 수 있어야 한다

### API 호출 함수

```typescript
// lib/ai/generate-tags.ts
import { getGeminiClient } from './gemini-client'

export async function generateTags(content: string): Promise<string[]> {
    if (content.length < 30) {
        throw new Error('노트가 너무 짧아 태그를 생성할 수 없습니다.')
    }

    const client = getGeminiClient()
    
    const prompt = `
다음 노트 내용을 분석하여 관련된 태그를 최대 6개까지 생성해주세요.
태그는 쉼표로 구분하여 작성하고, 각 태그는 간결한 단어나 짧은 구절로 작성해주세요.

규칙:
- 최대 6개까지만 생성
- 한글 또는 영문
- 중복 없이
- 관련성 높은 순서로

노트 내용:
${content}

태그만 출력해주세요 (예: 개발, AI, 프로그래밍, 기술, 블로그)
    `

    const result = await client.generateText({
        prompt,
        config: {
            maxOutputTokens: 100,
            temperature: 0.5,
        }
    })

    // 태그 파싱 및 정제
    const tags = result.text
        .split(',')
        .map(tag => tag.trim())
        .filter(tag => tag.length > 0 && tag.length < 20)
        .slice(0, 6)

    // 중복 제거
    return [...new Set(tags)]
}
```

### Server Action

```typescript
// app/notes/[id]/actions.ts
'use server'

import { generateTags } from '@/lib/ai/generate-tags'
import { db } from '@/lib/db/connection'
import { notes } from '@/lib/db/schema/notes'
import { eq } from 'drizzle-orm'
import { createClient } from '@/lib/supabase/server'

export async function generateNoteTags(noteId: string) {
    try {
        const supabase = await createClient()
        const { data: { user } } = await supabase.auth.getUser()
        
        if (!user) {
            return { success: false, error: '로그인이 필요합니다.' }
        }

        // 노트 조회
        const [note] = await db
            .select()
            .from(notes)
            .where(eq(notes.id, noteId))
            .limit(1)

        if (!note || note.userId !== user.id) {
            return { success: false, error: '노트를 찾을 수 없습니다.' }
        }

        // 태그 생성
        const tagsArray = await generateTags(note.content || '')
        const tagsString = tagsArray.join(', ')

        // DB 업데이트
        await db
            .update(notes)
            .set({ 
                tags: tagsString,
                updatedAt: new Date()
            })
            .where(eq(notes.id, noteId))

        return { success: true, tags: tagsArray }
    } catch (error) {
        return {
            success: false,
            error: error instanceof Error ? error.message : '태그 생성 실패'
        }
    }
}

export async function updateNoteTags(noteId: string, tags: string[]) {
    try {
        const supabase = await createClient()
        const { data: { user } } = await supabase.auth.getUser()
        
        if (!user) {
            return { success: false, error: '로그인이 필요합니다.' }
        }

        const tagsString = tags.join(', ')

        await db
            .update(notes)
            .set({ 
                tags: tagsString,
                updatedAt: new Date()
            })
            .where(eq(notes.id, noteId))

        return { success: true }
    } catch (error) {
        return { success: false, error: '태그 업데이트 실패' }
    }
}
```

### UI 컴포넌트

```typescript
// components/notes/note-tags.tsx
'use client'

import { useState } from 'react'
import { Button } from '@/components/ui/button'
import { Badge } from '@/components/ui/badge'
import { X } from 'lucide-react'
import { generateNoteTags, updateNoteTags } from '@/app/notes/[id]/actions'

export function NoteTags({ 
    noteId, 
    initialTags 
}: { 
    noteId: string
    initialTags?: string | null
}) {
    const [tags, setTags] = useState<string[]>(
        initialTags ? initialTags.split(', ').filter(Boolean) : []
    )
    const [loading, setLoading] = useState(false)
    const [error, setError] = useState<string>()

    const handleGenerate = async () => {
        setLoading(true)
        setError(undefined)

        const result = await generateNoteTags(noteId)

        setLoading(false)

        if (result.success && result.tags) {
            setTags(result.tags)
        } else {
            setError(result.error)
        }
    }

    const handleRemoveTag = async (tagToRemove: string) => {
        const newTags = tags.filter(tag => tag !== tagToRemove)
        setTags(newTags)
        await updateNoteTags(noteId, newTags)
    }

    return (
        <div className="mb-4">
            <div className="flex items-center gap-2 mb-2">
                <h3 className="text-sm font-semibold">🏷️ 태그</h3>
                <Button 
                    onClick={handleGenerate} 
                    disabled={loading}
                    variant="outline"
                    size="sm"
                >
                    {loading ? '생성 중...' : '✨ AI 태그 생성'}
                </Button>
            </div>

            {tags.length > 0 && (
                <div className="flex flex-wrap gap-2">
                    {tags.map((tag, index) => (
                        <Badge 
                            key={index} 
                            variant="secondary"
                            className="flex items-center gap-1"
                        >
                            {tag}
                            <button
                                onClick={() => handleRemoveTag(tag)}
                                className="ml-1 hover:text-red-600"
                            >
                                <X className="h-3 w-3" />
                            </button>
                        </Badge>
                    ))}
                </div>
            )}

            {error && (
                <p className="text-sm text-red-600 mt-2">⚠️ {error}</p>
            )}
        </div>
    )
}
```
